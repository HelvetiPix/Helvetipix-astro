// --- Distance Haversine (km) ---
function distanceKm(lat1, lon1, lat2, lon2) {
const R = 6371;
const dLat = (lat2 - lat1) * Math.PI / 180;
const dLon = (lon2 - lon1) * Math.PI / 180;
const a = Math.sin(dLat/2) * Math.sin(dLat/2) +
Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
Math.sin(dLon/2) * Math.sin(dLon/2);
return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
}

// --- Estimation pollution lumineuse (0-10, 10 = ciel pur) ---
function estimerPollutionLumineuse(lat, lon) {
let pollutionMax = 0; // 0 = aucune pollution
for (const ville of VILLES) {
const dist = distanceKm(lat, lon, ville.lat, ville.lon);
// Rayon d'influence : plus la ville est grande, plus loin elle éclaire
const rayon = 10 + (ville.pop / 100000) * 15; // 10 à ~70 km
if (dist < rayon) {
// Intensité inverse de la distance
const intensite = 10 * (1 - dist / rayon) * (0.5 + ville.pop / 200000);
pollutionMax = Math.max(pollutionMax, intensite);
}
}
// Score : 10 = pas de pollution, 0 = très pollué
return Math.round(Math.max(0, 10 - pollutionMax) * 10) / 10;
}

// --- Score météo (0-10) ---
function scoreMeteo(cloudCover, precipitation, visibility) {
let score = 10;
// Nuages : impact majeur
score -= (cloudCover / 100) * 6;
// Pluie : rédhibitoire
if (precipitation > 0.5) score -= 3;
else if (precipitation > 0.1) score -= 1.5;
// Visibilité : impact modéré
if (visibility !== null && visibility !== undefined) {
const visKm = visibility / 1000;
if (visKm < 5) score -= 2;
else if (visKm < 10) score -= 1;
}
return Math.max(0, Math.min(10, Math.round(score * 10) / 10));
}

// --- Score global astro (0-10) ---
function scoreAstroGlobal(meteo, pollution) {
// Météo = 60% du score, pollution = 40%
return Math.round((meteo * 0.6 + pollution * 0.4) * 10) / 10;
}

// --- Récupérer météo ---
async function fetchWeather(lat, lon) {
const url = https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&current=cloud_cover,precipitation,visibility&timezone=auto;
try {
const response = await fetch(url);
const data = await response.json();
if (data.error) throw new Error(data.reason);
return {
cloudCover: data.current.cloud_cover,
precipitation: data.current.precipitation || 0,
visibility: data.current.visibility
};
} catch (error) {
console.error("Erreur météo:", error);
return null;
}
}
